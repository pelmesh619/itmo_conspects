Но вы можете посмотреть рандомное фото котика:

<script>
document.addEventListener('DOMContentLoaded', function() {
    const imgElement = document.querySelector('#randomCatImage');

    if (!imgElement) {
        console.error('Тег <img id="randomCatImage"> не найден на странице');
        return;
    }

    const apiUrl = 'https://api.thecatapi.com/v1/images/search';

    fetch(apiUrl)
        .then(response => {
            if (!response.ok) {
                throw new Error(`Ошибка HTTP: ${response.status}`);
            }
            return response.json();
        })
        .then(data => {
            const imageUrl = data[0].url;
            imgElement.src = imageUrl;
            imgElement.alt = 'Случайный кот';
        })
        .catch(error => {
            console.error('Ошибка при загрузке кота:', error);
        });
});
</script>
